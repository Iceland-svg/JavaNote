
实现ResponseBodyAdvice接口，重写beforeBodyWrite方法

```java
@ControllerAdvice  
@Slf4j  
public class ResponseAdvice implements ResponseBodyAdvice {  
    @Autowired  
    private ObjectMapper objectMapper;  
    @Override  
    public boolean supports(MethodParameter returnType, Class converterType) {  
        return false;  
    }  
  //这个注解自动包上try,catch
    @SneakyThrows  
    @Nullable    @Override    public Object beforeBodyWrite(@Nullable Object body, MethodParameter returnType, MediaType selectedContentType, Class selectedConverterType, ServerHttpRequest request, ServerHttpResponse response) {  
        log.info("执行beforeBodyWrite方法");  
        if(body instanceof Result<?>){  
            return body;  
        }  
        if(body instanceof String){  
            return objectMapper.writeValueAsString(Result.success(body));  
        }  
        return Result.success(body);  
    }  
}
```
