1 在排序数组中查找元素的第一个和最后一个位置
https://leetcode.cn/problems/find-first-and-last-position-of-element-in-sorted-array/description/
1 暴力解
```java
class Solution {

    public int[] searchRange(int[] nums, int target) {

        int[] ret = new int[2];

        int[] ret1 = {-1,-1};

         int begin = -1 , end = -1;

         if(nums.length == 0) return ret1;

         for(int i = 0; i < nums.length; i++){

            if(nums[i] == target){

                begin = i;

                end = i;

                ret[0] = begin;

                break;

            }

         }

         if(begin == -1) return ret1;

         if(begin == nums.length-1) {

            ret[1] = begin;

            return ret;

         }

         for(int j = begin; j < nums.length-1; j++){

            if(nums[j+1] == target){

                end++;

                ret[1] = end;

            }else{

                ret[1] = end;

                return ret;

            }

         }

         return ret;

    }

}
```
